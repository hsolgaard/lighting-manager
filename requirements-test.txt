# release if you want to test against a newer HA core.
pytest-homeassistant-custom-component==0.13.108

# Pins around a real environment issue, not a Lighting Manager one:
# HA's http component imports hass_nabucasa -> acme, and acme 2.8.0
# uses the pre-2.0 josepy API (josepy.ComparableX509), which josepy>=2.0
# removed. Without this pin, any test that spins up hass.http (the
# websocket-API tests) fails with AttributeError on import, nothing to
# do with this integration's own code.
josepy==1.15.0

# Home Assistant's zoneinfo-based time handling needs the IANA tzdata
# database; not every base OS/container ships it.
tzdata
